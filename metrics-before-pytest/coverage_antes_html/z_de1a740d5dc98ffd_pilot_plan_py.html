27def schedule() -> list[dict]: 

28 """Canary occupies 16 of the 48 slots, never 16 additional slots.""" 

29 ordered = [b for b in BRIEFS if b[0] in CANARY] + [b for b in BRIEFS if b[0] not in CANARY] 

30 rows = [] 

31 for index, (brief, prompt) in enumerate(ordered): 

32 for take in range(1, TAKES + 1): 

33 # Both take strata and each phase balance which arm leads; each 

34 # brief also reverses its leader for the second take. 

35 arms = ARMS if (index + take) % 2 else ARMS[::-1] 

36 for arm in arms: 

37 rows.append({"slot": f"{brief}-{arm}-{take}", "brief_id": brief, "brief": prompt, 

38 "arm": arm, "take": take, "phase": "canary" if brief in CANARY else "remainder", 

39 "status": "planned"}) 

40 return rows

43def cost_ceiling(per_attempt: Decimal, budget: Decimal) -> dict: 

44 """Conservative full-run reservation in one user-selected currency/unit. 

45 

46 The operator must supply a verified all-in upper bound, not an average price. 

47 This arithmetic neither reserves funds nor authorizes execution.

49 for amount in (per_attempt, budget): 

50 if not isinstance(amount, Decimal) or not amount.is_finite() or amount <= 0: 

51 raise ValueError("cost bounds must be finite positive Decimal values")

53 context.prec = max(28, len(per_attempt.as_tuple().digits) + 2)
