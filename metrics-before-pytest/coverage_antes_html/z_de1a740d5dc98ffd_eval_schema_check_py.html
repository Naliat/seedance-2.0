154def main() -> int: 

155 parser = argparse.ArgumentParser() 

156 parser.add_argument("repo", nargs="?", default=".") 

157 args = parser.parse_args() 

158 

159 root = Path(args.repo).resolve() 

160 path = root / "evals" / "evals.json" 

161 if not path.exists(): 

162 print("Missing evals/evals.json") 

163 return 1 

164 

165 try: 

166 data = load_json(path, expected_type=dict, root=root) 

167 except Exception as exc: 

168 print(diagnostic_text(f"Invalid JSON: {exc}")) 

169 return 1 

170 

171 cases = data.get("cases") 

172 if not isinstance(cases, list) or len(cases) < 16: 

173 print("evals/evals.json must contain at least 16 cases") 

174 return 1 

175 

176 ids = set() 

177 errors = [] 

178 for i, case in enumerate(cases): 

179 if not isinstance(case, dict): 

180 errors.append(f"case {i} is not an object") 

181 continue 

182 for field in REQUIRED: 

183 if field not in case: 

184 errors.append(f"case {i} missing {field}") 

185 cid = case.get("id") 

186 if cid in ids: 

187 errors.append(f"duplicate id: {cid}") 

188 ids.add(cid) 

189 if not isinstance(case.get("assertions"), list) or len(case.get("assertions", [])) < 2: 

190 errors.append(f"case {cid} needs at least two assertions") 

191 if not isinstance(case.get("skills_expected_to_activate"), list) or not case.get("skills_expected_to_activate"): 

192 errors.append(f"case {cid} needs skills_expected_to_activate") 

193 

194 missing_ids = REQUIRED_IDS - ids 

195 if missing_ids: 

196 errors.append("missing required eval ids: " + ", ".join(sorted(missing_ids))) 

197 

198 if errors: 

199 print("Eval schema errors:") 

200 for error in errors: 

201 print(diagnostic_text(f"- {error}")) 

202 return 1 

203 

204 print(diagnostic_text(f"Eval schema passed: {len(cases)} cases.")) 

205 return 0
